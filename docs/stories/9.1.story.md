### API Specifications

**Message Protocol for ModifierType System Operations** [Source: architecture/tech-stack.md#process-communication-protocol]:
```lua
-- Incoming Message Structure for Modifier Operations
msg = {
    Action = "ProcessLogic",
    Data = {
        operation = "getModifierInfo", -- or "calculateHeldModifierEffects", "validateModifierUsage", "processModifierConsumption"
        gameState = {
            player = { 
                inventory = {
                    modifiers = {}, -- Player's modifier inventory (items/held items)
                    money = 0,  -- Currency for modifier purchasing
                },
                party = {} -- Pokemon party for held modifier processing
            },
            battle = { -- Current battle context for modifier effects
                playerParty = {},
                conditions = {},
                turnCommands = {}
            }
        },
        parameters = {
            modifierId = "POTION", -- Specific modifier for operation (matches ModifierTypes keys)
            targetPokemon = 0,  -- Pokemon index for modifier application
            modifierQuantity = 1,   -- Number of modifiers for consumption
            effectContext = "battle", -- battle, field, menu context
            modifierAction = "use" -- use, consume, apply, check
        }
    }
}

-- Response Message Structure
ao.send({
    Target = msg.From,
    Action = "SaveState", -- or "Error" for invalid operations
    Success = "true",
    Operation = "getModifierInfo", 
    Data = {
        modifierInfo = {
            modifierId = "POTION", -- Matches ModifierTypes key
            name = "Potion",
            description = "Heals a Pokemon by 20 HP",
            category = "MEDICINE", -- From ModifierType classification
            usageContext = "battle",
            effectType = "healing"
        },
        modifierEffects = {
            healingAmount = 20,
            targetType = "single",
            usageRestrictions = {"in_battle", "low_hp"}
        },
        heldModifierEffects = {
            statModifiers = {attack = 1.1, defense = 1.0}, -- From PokemonHeldItemModifier
            typeEffectiveness = {},
            specialEffects = []
        }
    },
    GameState = json.encode(updated_gamestate)
})
```
