                battleData = { -- Current battle context for fusion calculations
                    statModifiers = {}, -- Battle stat modifiers
                    statusEffects = {}, -- Active status effects
                    activeAbilities = {} -- Currently active abilities
                }
            },
            battle = { -- Current battle context for fusion determinism
                battleSeed = "12345", -- Seed for deterministic fusion battle calculations

                battleType = "WILD" -- Battle type affecting fusion AI behavior

        parameters = {

        }
    }
}
