#### Type Effectiveness Matrix (18×18 = 324 entries)
[Source: src/data/type.ts:5-267]
```lua
-- Embed as nested Lua table for efficient lookups

local TYPE_CHART = {
    [0] = { -- NORMAL defending
        [1] = 2,   -- FIGHTING attacks (super effective)
        [7] = 0,   -- GHOST attacks (immune)
        -- Unlisted combinations default to 1.0 (normal effectiveness)
    },
    [1] = { -- FIGHTING defending
        [2] = 2,   -- FLYING attacks (super effective)
        [13] = 2,  -- PSYCHIC attacks (super effective)
        [17] = 2,  -- FAIRY attacks (super effective)

        [15] = 0.5 -- DARK resists
    },
    -- ... (repeat for all 18 types: 0-17)

-- Default lookup with fallback to 1.0
local function getTypeMultiplier(attackType, defenseType)
    return TYPE_CHART[defenseType] and TYPE_CHART[defenseType][attackType] or 1.0
end
```

**Ability Integration Pattern**:
```lua
local function getMoveEffectiveness(moveType, targetTypes, targetAbility)
    -- 1. Calculate base type effectiveness from type chart
    local effectiveness = 1.0
    for _, defType in ipairs(targetTypes) do
        effectiveness = effectiveness * getTypeMultiplier(moveType, defType)
    end

    -- 2. Apply ability modifiers
    if targetAbility == 26 and moveType == 4 then -- Levitate vs Ground
        effectiveness = 0
    elseif targetAbility == 18 and moveType == 9 then -- Flash Fire vs Fire
        effectiveness = 0
    elseif targetAbility == 10 and moveType == 12 then -- Volt Absorb vs Electric
        effectiveness = 0
    -- ... (repeat for all 11 abilities)
    end

    return effectiveness
end
```

#### Dual-Type Calculation Algorithm
[Source: Implied by Pokemon mechanics, validated against src/data/type.ts behavior]
```lua
-- Multiply individual type effectiveness multipliers
local function calculateDualTypeEffectiveness(moveType, type1, type2)
    local eff1 = getTypeMultiplier(moveType, type1)
    local eff2 = type2 and getTypeMultiplier(moveType, type2) or 1.0
    return eff1 * eff2
end

-- Example: Electric move vs Water/Flying (Gyarados)
-- Electric vs Water: 2× (super effective)
-- Electric vs Flying: 2× (super effective)
-- Total: 2 × 2 = 4× (double weakness)
```

```lua
local function hasStab(pokemonTypes, moveType)
```
