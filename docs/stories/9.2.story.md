### API Specifications

**Message Protocol for Berry System Operations** [Source: architecture/tech-stack.md#process-communication-protocol]:
```lua
-- Incoming Message Structure for Berry Operations
msg = {
    Action = "ProcessLogic",
    Data = {
        operation = "evaluateBerryTrigger", -- or "consumeBerry", "restoreBerry", "getBerryInfo"
        gameState = {
            player = { 
                inventory = {
                    berries = {}, -- Player's berry inventory with quantities
                    heldBerries = {} -- Currently held berries on party Pokemon
                },
                party = {} -- Pokemon party for berry effect processing
            },
            battle = { -- Current battle context for berry triggers
                playerParty = {},
                conditions = {},
                turnCommands = {},
                currentPhase = "", -- Battle phase for timing validation
                eventSequence = {} -- Event ordering for trigger evaluation
            }
        },
        parameters = {
            berryId = "SITRUS_BERRY", -- Specific berry for operation (matches BerryModifierTypes keys)
            targetPokemon = 0,  -- Pokemon index for berry application
            triggerContext = "hp_below_50", -- Trigger condition context
            berryAction = "consume" -- consume, restore, evaluate, check
        }
    }
}

-- Response Message Structure
ao.send({
    Target = msg.From,
    Action = "SaveState", -- or "Error" for invalid operations
    Success = "true",
    Operation = "evaluateBerryTrigger", 
    Data = {
        berryInfo = {
            berryId = "SITRUS_BERRY", -- Matches BerryModifierTypes key
            name = "Sitrus Berry",
            description = "Heals 25% HP when HP falls below 50%",
            category = "BERRY", -- From BerryModifierType classification
            triggerCondition = "hp_below_50",
            effectType = "conditional_heal"
        },
        triggerEvaluation = {
            triggered = true,
            conditionMet = "hp_below_50",
            targetPokemon = 0,
            timing = "post_damage"
        },
        berryEffects = {
            healingAmount = 25, -- Percentage or fixed amount
            statusCured = nil, -- Status condition removed (if applicable)
            ppRestored = 0, -- PP amount restored (if applicable)
            statModifications = {} -- Stat changes from berry
        }
    },
    GameState = json.encode(updated_gamestate)
})
```
