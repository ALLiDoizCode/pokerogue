### API Specifications

**Message Protocol for Economic System Operations** [Source: architecture/tech-stack.md#process-communication-protocol]:
```lua
-- Incoming Message Structure for Economic Operations
msg = {
    Action = "ProcessLogic",
    Data = {
        operation = "generateShop", -- or "purchaseItem", "rerollShop", "calculateCost", "processReward"
        gameState = {
            player = { 
                inventory = {
                    money = 0, -- Current player money for purchase validation
                    items = {}, -- Current item inventory for purchase constraints
                    modifiers = {} -- Current modifiers affecting economic events
                },
                progression = {
                    wave = 1, -- Current wave for pricing and availability calculations
                    unlocks = {} -- Economic progression unlocks and milestones
                }
            },
            battle = { -- Current battle context for economic events
                waveIndex = 1, -- Wave progression for shop generation
                conditions = {}, -- Battle conditions affecting economic rewards
                rewardMultipliers = {} -- Economic multipliers from battle performance
            }
        },
        parameters = {
            shopOperation = "generate", -- generate, purchase, reroll, validate
            itemId = "POTION", -- Specific item for purchase operations (matches ModifierType keys)
            rerollCount = 0, -- Current reroll count for cost calculation
            lockTiers = false, -- Whether tier locking is enabled for reroll operations
            purchaseAmount = 1, -- Quantity for bulk purchase operations
            economicEvent = "battle_victory" -- Economic event type for reward processing
        }
    }
}

-- Response Message Structure  
ao.send({
    Target = msg.From,
    Action = "SaveState", -- or "Error" for invalid operations
    Success = "true",
    Operation = "generateShop",
    Data = {
        shopInventory = {
            items = {
                {
                    itemId = "POTION", -- Matches ModifierType key
                    name = "Potion",
                    description = "Heals 20 HP",
                    tier = "COMMON", -- From ModifierTier classification
                    cost = 75, -- Calculated cost based on wave and modifiers
                    available = true, -- Availability based on progression and constraints
                    category = "HEALING" -- Item category for shop organization
                }
            },
            rerollCost = 150, -- Current reroll cost based on wave and count
            tiersLocked = false, -- Whether tier locking is active
            waveIndex = 1 -- Wave context for shop generation
        },
        economicState = {
            playerMoney = 500, -- Current player money after any transactions
            transactionValid = true, -- Whether requested transaction is valid
            costCalculation = 75, -- Detailed cost breakdown for requested operation
            availabilityReason = "wave_progression" -- Reason for item availability status
        }
    },
    GameState = json.encode(updated_gamestate)
})
```
